def get_students_ajax(request):
    course_id = request.GET.get('course_id')
    semester_id = request.GET.get('semester_id')
    students = StudentProfile.objects.filter(course_id=course_id, semester_id=semester_id).select_related('user')
    student_list = [{'id': s.id, 'name': f"{s.roll_number} - {s.user.get_full_name()}"} for s in students]
    return JsonResponse(student_list, safe=False)
